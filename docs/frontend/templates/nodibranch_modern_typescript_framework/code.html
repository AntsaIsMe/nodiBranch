Familiar patterns. Exponential velocity.

Transitioning from PHP Symfony or NestJS? Nodibranch feels instantly organic with zero decorator black magic.

user.controller.ts
Nodibranch v1.0
// Pure TypeScript • Expressive Autowiring

import { Controller, Get, Inject } from 'nodibranch';

import { UserRepository } from './user.repository';


@Controller('/api/users')

export class UserController {

constructor(

@Inject() private readonly users: UserRepository

) {}


@Get()

async list() {

return await this.users.findAll();

}

}
